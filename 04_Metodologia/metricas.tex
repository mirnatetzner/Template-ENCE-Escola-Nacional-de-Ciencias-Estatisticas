\section{Métricas}\label{metricas}

% exemplos de equação

\begin{equation}\label{eq:medias_erro}
\bar{Q}_{bl} = \frac{1}{B} \sum_{b=1}^{B} {Q}_{bl},
\end{equation}
em que $\bar{Q}_{bl}$ representa a estimativa obtida na $b$-ésima repetição de \textit{bootstrap} no cenário $l$ e método de imputação específico.

\begin{equation}\label{eq:erro_bruto}
\text{Erro Bruto} = \bar{Q}_{bl} - Q.
\end{equation}
